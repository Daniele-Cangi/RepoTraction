def direct_geometry(codes, clocks, positions):
    angles, records = [], []
    for values, clock, xyz in zip(codes, clocks, positions, strict=True):
        result = study.solve(values-clock, xyz, np.eye(len(xyz))*400., atmosphere=False)
        best = min(b['cost'] for b in result['branches'])
        branches = [{**b, 'exterior': np.linalg.norm(b['q'][:3]) > study.dynamics.R_EARTH}
                    for b in result['branches']]
        retained = [b for b in branches if b['exterior'] and b['cost']-best <= CHI95]
        records.append({'branches': branches, 'eligible_branch_count': len(retained),
                        'cost_gap_threshold': CHI95})
        if len(retained) != 1:
            return {'status': 'RF_GEOMETRY_AMBIGUOUS', 'snapshots': records}, []
        angles.append([study.troposphere(s, np.asarray(retained[0]['q'][:3]))[1] for s in xyz])
    elevations = np.asarray(angles)
    selected = np.flatnonzero(np.min(elevations, axis=0) >= 10.)
    return {'status': 'RF_MASK_EVALUATED', 'snapshots': records, 'elevation_deg': elevations}, selected
