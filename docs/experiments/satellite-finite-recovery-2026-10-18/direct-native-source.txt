def make_problem(q,case,raw_covariance,data,weather):
    check_time()
    net=study.network_clocks.fit(case['reference_rows'],raw_covariance=raw_covariance)
    profile,w=study.source_geometry.profile_sources(net,q,study.source_geometry.FAMILIES)
    assert profile['resolved_rank']==case['source_profile']['resolved_rank']
    u,s,v=np.linalg.svd(w['design'],full_matrices=False)
    rank=profile['resolved_rank']
    provider,corrections=study.baseline.subset_products(data,[r for r in data['plan']['references'] if r!=q],q)
    codes={(station,e['time_s'],ref):value for station,epochs in data['observations'].items() for e in epochs
           for ref,value in e['if_code_m'].items() if ref!=q and ref in data['plan']['references']}
    mjd=(study.datetime.fromisoformat(data['plan']['date_gpst'])-study.datetime(1858,11,17)).days
    contexts={}
    for b in net['blocks']:
        for i,r in enumerate(b['rows']):
            if i%64==0:check_time()
            station,ref,t=r['station'],r['reference'],r['time_s']
            assert ref!=q
            xyz=np.asarray(data['positions'][station][int((t-36000)/30)],float)
            context=study.reference_geometry.StableSourceRay(provider,weather,mjd,ref,
                float(codes[station,t,ref])+corrections[t,ref]['if_code_change_m'],t,xyz,r['reference_clock_centre_m'])
            contexts[station,ref,t]=context
    # The estimator gets only allowed geometry, operators and observations.
    # Neither case/physical_probes nor any source truth is captured by forward.
    basis=v[:rank].T.copy();ur=u[:,:rank].copy();sr=s[:rank].copy();columns=w['columns']
    calls=[0]
    def forward(alpha):
        check_time();calls[0]+=1
        split=study.source_geometry.split_probe(columns,basis@alpha,q)
        blocks=[]
        for b in net['blocks']:
            values=[]
            for i,r in enumerate(b['rows']):
                if i%64==0:check_time()
                station,ref,t=r['station'],r['reference'],r['time_s']
                local=[split['wet'].get(station,0.),*split['orbit'].get(ref,[0.,0.,0.]),
                       *split['antenna'].get(ref,[0.,0.,0.]),split['bias'].get(station+'/'+ref,0.)]
                values.append(contexts[station,ref,t].increment(local))
            blocks.append(np.asarray(values).T.ravel())
        return study.source_geometry.profiled_increment(net,blocks)
    return {'basis':basis,'u':ur,'s':sr,'forward':forward,'calls':calls,
            'profile_observation':lambda blocks:study.source_geometry.profiled_increment(net,blocks)}

def estimate(problem,observed):
    alpha=problem['u'].T@observed/problem['s']
    first_calls=problem['calls'][0];history=[]
    if METHOD=='chord':
        for iteration in range(8):
            predicted=problem['forward'](alpha)
            residual=observed-predicted
            step=(problem['u'].T@residual)/problem['s']
            history.append({'iteration':iteration,'weighted_residual_norm':float(np.linalg.norm(residual)),
                            'step_l2_m':float(np.linalg.norm(step))})
            alpha=alpha+step
            if np.linalg.norm(step)<=1e-8*max(1.,np.linalg.norm(alpha)):
                return {'status':'CONVERGED','estimate':problem['basis']@alpha,'history':history,
                        'forward_calls':problem['calls'][0]-first_calls}
        return {'status':'ITERATION_LIMIT','estimate':problem['basis']@alpha,'history':history,
                'forward_calls':problem['calls'][0]-first_calls}
    result=least_squares(lambda z:problem['forward'](z)-observed,alpha,jac='3-point',
                         max_nfev=20,ftol=1e-10,xtol=1e-10,gtol=1e-12)
    return {'status':'CONVERGED' if result.success else 'ITERATION_LIMIT',
            'estimate':problem['basis']@result.x,'forward_calls':problem['calls'][0]-first_calls,
            'scipy_status':int(result.status),'scipy_nfev':int(result.nfev),'cost':float(result.cost)}
